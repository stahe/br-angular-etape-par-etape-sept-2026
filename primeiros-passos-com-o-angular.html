    // =========================================================================
// 01-first-application/src/app/app.ts - um primeiro componente
// -------------------------------------------------------------------------
// Um componente Angular é uma CLASSE TypeScript, decorada por @Component:
//   - selector    : a tag HTML que o representa (<app-root>);
//   - templateUrl : o arquivo do seu HTML (app.html);
//   - styleUrl    : o arquivo do seu CSS (app.css), que se aplica apenas
//                   a este componente.
// As propriedades e os métodos da classe podem ser usados no template.
// =========================================================================

import { Component, signal } from '@angular/core';

@Component({
  selector: 'app-root',
  templateUrl: './app.html',
  styleUrl: './app.css',
})
export class App {
  // dados "comuns": o template pode exibi-los
  protected readonly title = 'Welcome to Angular';
  protected readonly languages = ['TypeScript', 'HTML', 'CSS'];

  // signal(...): um dado REATIVO. Seu valor é lido ao CHAMÁ-LO
  // (counter()) e alterado com set() ou update(). O Angular então
  // atualiza AUTOMATICAMENTE os pontos da página que o exibem.
  protected readonly counter = signal(0);
  protected readonly time = signal(new Date().toLocaleTimeString());

  // um método chamado pelo botão do template
  protected increment(): void {
    this.counter.update((n) => n + 1);
    this.time.set(new Date().toLocaleTimeString());
  }
}
    • Linhas 14–18: o decorador [@Component] transforma a classe em um componente:
      • linha 15: seu seletor, a tag que o representa em um template ou no [index.html];
      • linha 16: seu arquivo de template, HTML;
      • linha 17: seu arquivo de estilo, CSS;
    • linhas 21–22: dois dados comuns. [protected]: são usados apenas pelo componente e por seu template; [readonly]: não serão substituídos;
    • linha 27: [signal(0)] cria um dado reativo com valor inicial 0. Um signal é uma função: você lê seu valor chamando-o ([this.counter()]) e o altera com [set(value)] ou [update(function)]. Quando ele muda, o [Angular] atualiza todos os locais da página que o exibem;
    • linhas 31–34: um método que modifica dois signals. Ele não altera a página: o [Angular] a atualizará. [update((n) => n + 1)] calcula o novo valor a partir do antigo.

    // =========================================================================
// 02-signals/src/app/app.ts - signal, computed, effect, linkedSignal
// -------------------------------------------------------------------------
// Uma pequena calculadora de preços com IVA:
//   - signal       : um valor reativo, lido ao chamá-lo e
//                    modificado com set() / update();
//   - computed     : um valor CALCULADO a partir de outros signals,
//                    recalculado automaticamente quando eles mudam;
//   - effect       : código executado a cada mudança dos signals que ele
//                    lê (um "efeito colateral");
//   - linkedSignal : um signal gravável, mas REINICIADO para um valor calculado
//                    quando outro signal muda.
// =========================================================================

import { Component, computed, effect, linkedSignal, signal, untracked } from '@angular/core';
// FormsModule fornece [(ngModel)], que vincula um campo a um signal (veja o exemplo 05)
import { FormsModule } from '@angular/forms';

@Component({
  selector: 'app-root',
  imports: [FormsModule],
  templateUrl: './app.html',
})
export class App {
  protected readonly label = signal('Pen');
  protected readonly priceExclVat = signal(2.5);
  protected readonly vatRate = signal(20); // em %

  // linkedSignal: a quantidade volta a 1 a cada vez que o item muda,
  // mas o usuário pode modificá-la nesse meio tempo
  protected readonly quantity = linkedSignal({ source: this.label, computation: () => 1 });

  // computed: nunca atribuído diretamente, sempre recalculado. O Angular sabe
  // de quais signals ele depende (os que ele LÊ): recalcula-o quando um
  // deles muda, e somente nesse caso
  protected readonly totalExclVat = computed(() => this.quantity() * this.priceExclVat());
  protected readonly vatAmount = computed(() => (this.totalExclVat() * this.vatRate()) / 100);
  protected readonly totalInclVat = computed(() => this.totalExclVat() + this.vatAmount());

  // um formatador de valores em euros, estilo inglês (€1,234.50)
  private readonly euros = new Intl.NumberFormat('en-US', { style: 'currency', currency: 'EUR' });

  // o histórico dos totais com IVA, preenchido pelo effect abaixo
  protected readonly history = signal<string[]>([]);
  private oldTotal: number | null = null;

  constructor() {
    // effect: executado uma primeira vez, depois a cada mudança de
    // totalInclVat (o único signal que ele lê)
    effect(() => {
      const total = this.totalInclVat();
      document.title = `Incl. VAT: ${this.format(total)}`;
      if (this.oldTotal !== null) {
        const line = `${this.format(this.oldTotal)} -> ${this.format(total)}`;
        // untracked: modifica history SEM torná-lo uma dependência do effect
        untracked(() => this.history.update((h) => [line, ...h].slice(0, 5)));
      }
      this.oldTotal = total;
    });
  }

  protected format(amount: number): string {
    return this.euros.format(amount);
  }

  protected reset(): void {
    this.label.set('Pen');
    this.priceExclVat.set(2.5);
    this.vatRate.set(20);
    this.quantity.set(1);
  }
}
    • linha 15: as funções de reatividade, todas fornecidas por [@angular/core];
    • Linhas 21 e 17: o template usa [[(ngModel)]], que vem do módulo [FormsModule]. Um componente deve importar (array [imports]) tudo o que seu template usa: outros componentes, diretivas e pipes. O compilador reporta um erro se isso for omitido;
    • linhas 25–27: três signals, [signal(initial value)]. O tipo deles é inferido a partir do valor inicial: [signal('Pen')] é um signal de string, [signal(2.5)] é um signal numérico;
    • linha 31: um [linkedSignal] é um signal modificável (pode receber escrita, como um signal comum), mas está vinculado a outro signal, sua origem: sempre que a origem muda (neste caso, a descrição do item), seu valor é redefinido para o valor calculado por [computation] (neste caso, 1). Quando o item muda, a quantidade volta a 1;
    • linhas 36–38: um [computed] é um valor calculado a partir de outros signals. Ele nunca recebe escrita; é lido como um signal ([this.totalExclVat()]). import { Component, signal } from '@angular/core';

interface Task {
  id: number;
  label: string;
  done: boolean;
}

@Component({
  selector: 'app-root',
  templateUrl: './app.html',
  styles: `
    .area {
      background: #e9f2ff;
      height: 80px;
      cursor: crosshair;
    }
  `,
})
export class App {
  protected readonly tasks = signal<Task[]>([
    { id: 1, label: 'Install Node.js', done: true },
    { id: 2, label: 'Read the chapter on events', done: false },
  ]);
  protected readonly log = signal<string[]>([]);
  private nextId = 3;

  private trace(message: string): void {
    this.log.update((j) => [`${new Date().toLocaleTimeString()} ${message}`, ...j]);
  }

  // chamado por (submit): $event.preventDefault() impede o navegador
  // de enviar o formulário ao servidor (e de recarregar a página)
  protected add(event: SubmitEvent, field: HTMLInputElement): void {
    event.preventDefault();
    const label = field.value.trim();
    if (label === '') {
      return;
    }
    // um signal contendo um array: nós o SUBSTITUÍMOS por um novo array
    this.tasks.update((t) => [...t, { id: this.nextId++, label, done: false }]);
    this.trace(`added "${label}"`);
    field.value = '';
    field.focus();
  }

  protected toggle(task: Task, done: boolean): void {
    this.tasks.update((t) => t.map((x) => (x.id === task.id ? { ...x, done } : x)));
    this.trace(`"${task.label}": ${done ? 'done' : 'to do'}`);
  }

  protected remove(task: Task): void {
    this.tasks.update((t) => t.filter((x) => x.id !== task.id));
    this.trace(`deleted "${task.label}"`);
  }

  // $event: o evento nativo (aqui um MouseEvent), com suas coordenadas
  protected position(event: MouseEvent): void {
    this.trace(`click at (${event.offsetX}, ${event.offsetY})`);
  }

// 06-signal-forms/src/app/app.ts - signal forms
// -------------------------------------------------------------------------
// A forma moderna (Angular 22) de escrever um formulário:
//   1. o MODELO: um signal que contém um objeto, com uma propriedade por
//      campo (o formulário lê e escreve nesse signal);
//   2. form(model, schema): constrói uma ÁRVORE DE CAMPOS (FieldTree), que
//      tem a mesma forma do modelo: f.lastName, f.age, f.newsletter... O
//      schema (opcional) descreve as regras dos campos (veja o exemplo 07);
//   3. [formField]="f.lastName" vincula um campo HTML a um campo da árvore.
// Cada campo tem um ESTADO, obtido ao chamá-lo: f.lastName() fornece seu valor
// (value()), se foi tocado (touched()), modificado (dirty()), se está
// desabilitado (disabled())...
// =========================================================================

import { Component, signal } from '@angular/core';
import { JsonPipe } from '@angular/common';
import { disabled, form, FormField } from '@angular/forms/signals';

// a forma dos dados do formulário
interface Profile {
  title: string;
  firstName: string;
  lastName: string;
  age: number;
  newsletter: boolean;
  email: string;
  country: string;
  birthDate: string;
  presentation: string;
}

const INITIAL: Profile = {
  title: 'Mme',
  firstName: 'Marie',
  lastName: 'PELISSIER',
  age: 30,
  newsletter: false,
  email: '',
  country: 'fr',
  birthDate: '',
  presentation: '',
};

@Component({
  selector: 'app-root',
  imports: [FormField, JsonPipe],
  templateUrl: './app.html',
})
export class App {
  protected readonly titles = ['Mr', 'Mme', 'Melle'];

  // 1. o modelo
  protected readonly model = signal<Profile>({ ...INITIAL });

  // 2. a árvore de campos; uma regra: o e-mail só pode ser modificado se a
  // caixa "newsletter" estiver marcada (valueOf lê o valor de OUTRO campo)
  protected readonly f = form(this.model, (p) => {
    disabled(p.email, ({ valueOf }) => !valueOf(p.newsletter));
  });

  // os campos cujo estado é exibido, na tabela à direita
  protected readonly fields = ['firstName', 'lastName', 'age', 'email'] as const;

  protected reset(): void {
    // o modelo volta ao seu valor inicial, e o estado dos campos é zerado
    this.model.set({ ...INITIAL });
    this.f().reset();
  }

  protected upperCase(): void {
    // modificando o MODELO por código: os campos acompanham
    this.model.update((m) => ({ ...m, lastName: m.lastName.toUpperCase(), firstName: m.firstName.toUpperCase() }));
  }
}

//
// Mesmas regras do servidor RdvMedecins (formulário "Create an account").
// ATENÇÃO: esta validação é apenas uma COMODIDADE para o usuário (uma
// resposta imediata, sem ida e volta ao servidor). Ela não protege
// nada: qualquer pessoa pode enviar ao servidor uma requisição montada à
// mão (curl...). O servidor, portanto, SEMPRE revalida tudo (veja o exemplo 25).

import { Component, signal } from '@angular/core';
import { form, FormField, maxLength, minLength, pattern, required, submit, validate } from '@angular/forms/signals';

interface Account {
  title: string;
  firstName: string;
  lastName: string;
  login: string;
  password: string;
  confirmation: string;
}

const TITLES = ['Mr', 'Mme', 'Melle'];
const LOGIN_FORMAT = /^[a-z0-9._-]{1,30}$/;
const PASSWORD_MIN = 6;

@Component({
  selector: 'app-root',
  imports: [FormField],
  templateUrl: './app.html',
})
export class App {
  protected readonly titles = TITLES;
  protected readonly model = signal<Account>({
    title: 'Mr',
    firstName: '',
    lastName: '',
    login: '',
    password: '',
    confirmation: '',
  });

  // o schema: p designa os CAMINHOS dos campos (p.lastName, p.login...)
  protected readonly f = form(this.model, (p) => {
    validate(p.title, ({ value }) =>
      TITLES.includes(value()) ? null : { kind: 'title', message: 'Choose a title from the list.' },
    );
    required(p.firstName, { message: 'The first name is required.' });
    maxLength(p.firstName, 20, { message: 'The first name must not exceed 20 characters.' });
    required(p.lastName, { message: 'The last name is required.' });
    maxLength(p.lastName, 20, { message: 'The last name must not exceed 20 characters.' });
    required(p.login, { message: 'The login is required.' });
    pattern(p.login, LOGIN_FORMAT, { message: 'Unaccented lower-case letters, digits, dot, hyphen, underscore.' });
    required(p.password, { message: 'The password is required.' });
    minLength(p.password, PASSWORD_MIN, { message: `At least ${PASSWORD_MIN} characters.` });
    // uma regra personalizada, que compara DOIS campos (valueOf lê o outro)
    validate(p.confirmation, ({ value, valueOf }) =>
      value() === valueOf(p.password)
        ? null
        : { kind: 'confirmation', message: 'The confirmation does not match the password.' },
    );
  });

  protected readonly submitAttempted = signal(false);
  protected readonly success = signal(false);

  // (submit) do formulário
  protected async send(event: SubmitEvent): Promise<void> {
    event.preventDefault();
    this.submitAttempted.set(true);
    // submit() marca todos os campos como touched (seus erros
    // aparecem) e só então executa a ação se tudo for válido
    const valid = await submit(this.f, async () => {
      // aqui, enviaríamos o modelo ao servidor (veja o exemplo 25)
      return null; // null: sem erro
    });
    this.success.set(valid);
  }
}

import { Component, inject, signal } from '@angular/core';
import { AbstractControl, FormBuilder, ReactiveFormsModule, ValidationErrors, Validators } from '@angular/forms';

const TITLES = ['Mr', 'Mme', 'Melle'];

// um validador personalizado se aplica ao GRUPO: ele compara dois de seus campos.
// Retorna null (sem erro) ou um objeto que descreve o erro.
function sameConfirmation(group: AbstractControl): ValidationErrors | null {
  const password = group.get('password')?.value;
  const confirmation = group.get('confirmation')?.value;
  return password === confirmation ? null : { confirmation: true };
}

@Component({
  selector: 'app-root',
  imports: [ReactiveFormsModule],
  templateUrl: './app.html',
})
export class App {
  protected readonly titles = TITLES;
  private readonly fb = inject(FormBuilder).nonNullable;

  // o grupo: { name: [valor inicial, validadores] }
  protected readonly form = this.fb.group(
    {
      title: ['Mr', [Validators.required]],
      firstName: ['', [Validators.required, Validators.maxLength(20)]],
      lastName: ['', [Validators.required, Validators.maxLength(20)]],
      login: ['', [Validators.required, Validators.pattern(/^[a-z0-9._-]{1,30}$/)]],
      password: ['', [Validators.required, Validators.minLength(6)]],
      confirmation: [''],
    },
    { validators: sameConfirmation },
  );

  protected readonly success = signal(false);
  protected readonly value = signal('');

  // um campo é sinalizado com erro se foi tocado e é inválido
  protected inError(name: string): boolean {
    const control = this.form.get(name);
    return !!control && control.touched && control.invalid;
  }

  // (ngSubmit): todos os campos passam a "touched", seus erros aparecem
  protected send(): void {
    this.form.markAllAsTouched();
    this.success.set(this.form.valid);
    // getRawValue(): o valor do grupo inteiro, tipado
    this.value.set(JSON.stringify(this.form.getRawValue()));
  }
}

// 09-pipes/src/app/pipes/time-range.pipe.ts - um pipe personalizado
// -------------------------------------------------------------------------
// Um pipe é uma classe decorada por @Pipe, cujo método transform()
// recebe o valor colocado ANTES da barra (e, eventualmente,
// parâmetros) e retorna o que deve ser exibido:
//     {{ slot | timeRange }}           -> "8:00 - 8:20"
//     {{ slot | timeRange: 'start' }}  -> "8:00"
// Um pipe é "puro" por padrão: o Angular só chama transform() novamente se o
// valor ou os parâmetros MUDARAM (mesmo objeto = mesmo resultado).
// =========================================================================

import { Pipe, PipeTransform } from '@angular/core';

export interface Slot {
  startHour: number;
  startMinute: number;
  endHour: number;
  endMinute: number;
}

// "8:05" em vez de "8:5"
function time(h: number, m: number): string {
  return `${h}:${String(m).padStart(2, '0')}`;
}

@Pipe({ name: 'timeRange' })
export class TimeRangePipe implements PipeTransform {
  transform(c: Slot, part: 'all' | 'start' | 'end' = 'all'): string {
    switch (part) {
      case 'start':
        return time(c.startHour, c.startMinute);
      case 'end':
        return time(c.endHour, c.endMinute);
      default:
        return `${time(c.startHour, c.startMinute)} - ${time(c.endHour, c.endMinute)}`;
    }
  }
}
    • linha 27: o nome do pipe, aquele escrito nos templates;
    • linha 29: [transform(value, part)] retorna o que deve ser exibido. O parâmetro é opcional (valor padrão ['all']): [{{ slot | timeRange }}] exibe [8:00 - 8:20], [{{ slot | timeRange: 'start' }}] exibe [8:00].

    Um pipe é puro por padrão: [Angular] só chama [transform] se o valor ou um dos parâmetros mudou (um objeto diferente, uma string diferente). É isso que o torna mais eficiente do que um método do componente chamado dentro do template (como [format(...)] no exemplo 02), que é executado toda vez que o componente é verificado.

import { Pipe, PipeTransform } from '@angular/core';

@Pipe({ name: 'initials' })
export class InitialsPipe implements PipeTransform {
  transform(fullName: string): string {
    return fullName
      .split(/\s+/)
      .filter((word) => word !== '')
      .map((word) => word[0].toUpperCase())
      .join('');
  }
}
