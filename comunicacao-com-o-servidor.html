    8. Comunicação com o servidor

    Os exemplos deste capítulo consultam o servidor JSON [RdvMedecins]: ele deve estar em execução ([npm run start:dev] na pasta [rdvmedecins-nestjs-json]) na porta 8080, ao mesmo tempo que o servidor de desenvolvimento do [Angular] na porta 4200.

    8.1. Exemplo 22: uma primeira requisição HTTP

    8.1.1. O proxy do servidor de desenvolvimento

    A página de exemplo vem de http://localhost:4200 (o servidor de desenvolvimento do [Angular], iniciado por [ng serve]); a API está em http://localhost:8080. Para um navegador, essas são duas origens diferentes (a porta faz parte da origem). Ora, por motivos de segurança, um navegador não permite que uma página leia respostas de outra origem, a menos que o servidor o permita explicitamente (o mecanismo conhecido como CORS). E uma requisição a outra origem não envia cookies (e, portanto, não envia o token), a menos que o código o solicite ([withCredentials: true]) E que o servidor o aceite, novamente por meio do CORS.

    Em vez de abrir o servidor a outras origens, encaminhamos as requisições à API pelo servidor de desenvolvimento do [Angular]. Este é o proxy declarado no arquivo do workspace [proxy.conf.json]:

// 22-httpclient/src/app/app.ts - uma primeira requisição HTTP ao servidor
// -------------------------------------------------------------------------
// O servidor JSON do RdvMedecins deve estar em execução (porta 8080). Pedimos a ele
// um captcha: GET /api/captcha -> { "id": "...", "svg": "<svg ...>" }.
//
//   - HttpClient: o cliente HTTP do Angular (fornecido por padrão para toda a
//     aplicação). http.get<T>(url) retorna um OBSERVABLE (veja o exemplo
//     19): a requisição só é enviada quando nos INSCREVEMOS nele (subscribe);
//     a resposta JSON chega já convertida em um objeto, do tipo T anunciado;
//   - o endereço é RELATIVO (/api/captcha): a requisição vai para o
//     servidor da página, o do Angular (porta 4200), que a repassa ao
//     servidor JSON (porta 8080) graças ao seu proxy (veja proxy.conf.json);
//   - três estados a exibir: carregando, sucesso, erro.
// =========================================================================

import { Component, computed, inject, signal } from '@angular/core';
import { HttpClient, HttpErrorResponse } from '@angular/common/http';
import { DomSanitizer, SafeHtml } from '@angular/platform-browser';

// o que GET /api/captcha retorna
interface Captcha {
  id: string;
  svg: string;
}

@Component({
  selector: 'app-root',
  templateUrl: './app.html',
  styles: `
    .captcha {
      display: inline-block;
      border: 1px solid #dee2e6;
      border-radius: 0.375rem;
    }
  `,
})
export class App {
  private readonly http = inject(HttpClient);
  private readonly sanitizer = inject(DomSanitizer);

  protected readonly captcha = signal<Captcha | null>(null);
  protected readonly loading = signal(false);
  protected readonly error = signal<string | null>(null);
  protected readonly requestCount = signal(0);

  // [innerHTML] insere uma string como HTML. Por segurança, o Angular
  // LIMPA esse HTML ("sanitization"): remove os scripts... e as
  // tags <svg>. bypassSecurityTrustHtml diz ao Angular: "este HTML é
  // seguro, não o limpe". Deve ser usado SOMENTE para conteúdo seguro: este SVG
  // é construído pelo NOSSO servidor. Com dados do usuário, seria
  // abrir a porta para a injeção de scripts ({{ }}, por outro lado, escapa tudo).
  protected readonly svg = computed<SafeHtml | null>(() => {
    const c = this.captcha();
    return c ? this.sanitizer.bypassSecurityTrustHtml(c.svg) : null;
  });

  constructor() {
    this.load(); // no início, carregamos um captcha
  }

  protected load(): void {
    this.loading.set(true);
    this.error.set(null);
    this.http.get<Captcha>('/api/captcha').subscribe({
      // next: a resposta (código 2xx), já convertida do JSON
      next: (captcha) => {
        this.captcha.set(captcha);
        this.requestCount.update((n) => n + 1);
        this.loading.set(false);
      },
      // error: uma resposta 4xx / 5xx, ou nenhuma resposta (status 0)
      error: (e: HttpErrorResponse) => {
        this.error.set(`the server answered ${e.status} ${e.statusText}`);
        this.captcha.set(null);
        this.loading.set(false);
      },
    });
  }
}

    • linhas 67–71: [next] recebe a resposta, que já foi convertida de JSON. [<Captcha>] informa ao [TypeScript]o formato do objeto: o compilador não tem como verificar o que o servidor enviará;
    • Linhas 73–77: [error] recebe um [HttpErrorResponse] para qualquer resposta 4xx ou 5xx, bem como quando nenhuma resposta é recebida (servidor fora do ar, rede desconectada: [status] vale então 0). Se o servidor JSON estiver fora do ar, o proxy responde com um erro 5xx;
    • linha 59: na inicialização, um CAPTCHA é carregado;

Nós o usamos SOMENTE para conteúdo seguro: este SVG é gerado pelo NOSSO servidor. // 23-login/src/app/app.ts - fazendo login no servidor RdvMedecins
// -------------------------------------------------------------------------
//   1. na partida: GET /api/auth/me - já estamos logados?
//      (depois de um F5, o cliente esqueceu tudo; o servidor lê o cookie);
//   2. login: POST /api/auth/login, corpo JSON { login, password,
//      captchaId, captcha }. Em caso de sucesso, o servidor define o
//      cookie httpOnly do token JWT: nosso código NUNCA o vê (tente
//      document.cookie no console: ele não está lá);
//   3. requisição protegida: GET /api/doctors. O navegador anexa o cookie
//      a ela por conta própria (mesma origem);
//   4. logout: POST /api/auth/logout - o servidor apaga o cookie.
//
// firstValueFrom(observable): uma PROMISE do primeiro valor do fluxo.
// Sequências de requisições podem então ser escritas com async / await,
// mais legível do que uma cascata de subscribe. Uma resposta de erro rejeita
// a promise com um HttpErrorResponse: um simples try / catch.
//
// Um cookie pertence a uma máquina (localhost) e a um caminho (/api): ele
// é portanto comum a TODOS os exemplos. Uma vez logado aqui, os exemplos
// 24 e 25 também ficam logados.
// =========================================================================

import { Component, computed, inject, signal } from '@angular/core';
import { HttpClient, HttpErrorResponse } from '@angular/common/http';
import { DomSanitizer } from '@angular/platform-browser';
import { FormsModule } from '@angular/forms';
import { firstValueFrom } from 'rxjs';

interface User {
  login: string;
  name: string;
  role: 'ADMIN' | 'DOCTOR' | 'USER';
}
interface Doctor {
  id: number;
  title: string;
  firstName: string;
  lastName: string;
}
interface Captcha {
  id: string;
  svg: string;
}
// a forma de TODOS os erros da API
interface ApiError {
  statusCode: number;
  key: string;
  params?: Record<string, string>;
}

// as poucas mensagens de que este exemplo precisa (o exemplo 25 usará o ngx-translate)
const MESSAGES: Record<string, string> = {
  'LOGIN.ERROR': 'Incorrect login or password.',
  'ERRORS.WRONG_CAPTCHA': 'The characters you typed do not match the image.',
  'ERRORS.TOO_MANY_ATTEMPTS': 'Too many failed attempts. Try again later.',
};

@Component({
  selector: 'app-root',
  imports: [FormsModule],
  templateUrl: './app.html',
})
export class App {
  private readonly http = inject(HttpClient);
  private readonly sanitizer = inject(DomSanitizer);

  protected readonly user = signal<User | null>(null);
  protected readonly doctors = signal<Doctor[]>([]);
  protected readonly captcha = signal<Captcha | null>(null);
  protected readonly svg = computed(() => this.sanitizer.bypassSecurityTrustHtml(this.captcha()?.svg ?? ''));
  protected readonly login = signal('admin');
  protected readonly password = signal('');
  protected readonly captchaAnswer = signal('');
  protected readonly error = signal<string | null>(null);
  protected readonly ready = signal(false); // a verificação inicial terminou?

  constructor() {
    this.checkSession().catch(() => {
      this.error.set('Is the RdvMedecins JSON server running (port 8080)?');
      this.ready.set(true);
    });
  }

  protected async loadCaptcha(): Promise<void> {
    this.captcha.set(await firstValueFrom(this.http.get<Captcha>('/api/captcha')));
    this.captchaAnswer.set('');
  }

  // 1. quem está logado? 200 -> o usuário; 401 -> ninguém
  private async checkSession(): Promise<void> {
    try {
      this.user.set(await firstValueFrom(this.http.get<User>('/api/auth/me')));
      await this.loadDoctors();
    } catch (e) {
      if (!(e instanceof HttpErrorResponse) || e.status !== 401) {
        throw e; // não é um 401: o servidor não responde
      }
      await this.loadCaptcha();
    }
    this.ready.set(true);
  }

  // 2. o login; post(url, body): o corpo é enviado como JSON
  protected async logIn(): Promise<void> {
    this.error.set(null);
    try {
      const u = await firstValueFrom(
        this.http.post<User>('/api/auth/login', {
          login: this.login(),
          password: this.password(),
          captchaId: this.captcha()?.id,
          captcha: this.captchaAnswer(),
        }),
      );
      this.user.set(u);
      this.password.set('');
      await this.loadDoctors();
    } catch (e) {
      // e.error: o corpo JSON da resposta de erro
      const apiError = (e as HttpErrorResponse).error as ApiError | null;
      const key = apiError?.key ?? 'ERRORS.SERVER_UNREACHABLE';
      this.error.set(`${(e as HttpErrorResponse).status} - ${MESSAGES[key] ?? key}`);
      await this.loadCaptcha(); // cada imagem é usada apenas uma vez
    }
  }

  // 3. uma requisição protegida
  private async loadDoctors(): Promise<void> {
    this.doctors.set(await firstValueFrom(this.http.get<Doctor[]>('/api/doctors')));
  }

  // 4. o logout
  protected async logOut(): Promise<void> {
    await firstValueFrom(this.http.post('/api/auth/logout', {}));
    this.user.set(null);
    this.doctors.set([]);
    await this.loadCaptcha();
  }
}

// common/api-error.ts - UMA classe para todos os erros da API
// -------------------------------------------------------------------------
// Todas as respostas de erro do servidor RdvMedecins têm a mesma forma:
//   { "statusCode": 409, "key": "ERRORS.LOGIN_TAKEN", "params": {...}, "fields": {...} }
// O HttpClient as reporta com um HttpErrorResponse, cuja propriedade error
// contém esse corpo JSON... ou outra coisa: null (nenhuma resposta), texto
// (uma página de erro do proxy quando o servidor JSON está parado). Convertemos
// tudo isso em UMA classe, ApiError, que carrega a chave de tradução
// enviada pelo servidor (e, se houver, seus parâmetros e
// o erro de cada campo).
// =========================================================================

import { HttpErrorResponse } from '@angular/common/http';

export class ApiError extends Error {
  constructor(
    public readonly statusCode: number, // 0: o servidor não respondeu
    public readonly key: string, // p. ex.: 'ERRORS.LOGIN_TAKEN'
    public readonly params: Record<string, string> = {},
    public readonly fields: Record<string, string> = {},
  ) {
    super(key);
    this.name = 'ApiError';
  }
}

// qualquer erro -> um ApiError
export function toApiError(e: unknown): ApiError {
  if (e instanceof ApiError) {
    return e;
  }
  if (!(e instanceof HttpErrorResponse)) {
    return new ApiError(0, 'ERRORS.SERVER_ERROR');
  }
  // o corpo JSON do erro, se existir e tiver a forma correta
  const body = (typeof e.error === 'object' && e.error !== null ? e.error : {}) as {
    key?: string;
    params?: Record<string, string>;
    fields?: Record<string, string>;
  };
  // sem key: não foi o nosso servidor que respondeu (proxy, rede)
  const key = body.key ?? (e.status === 0 || e.status >= 500 ? 'ERRORS.SERVER_UNREACHABLE' : 'ERRORS.SERVER_ERROR');
  return new ApiError(e.status, key, body.params, body.fields);
}

function tomorrow(): string {
  const d = new Date(Date.now() + 24 * 3600 * 1000);
  return `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, '0')}-${String(d.getDate()).padStart(2, '0')}`;
}

// 25-server-errors/src/app/app.ts - exibindo os erros do SERVIDOR
// -------------------------------------------------------------------------
// O gerenciamento dos clientes (conta admin). Este formulário
// DELIBERADAMENTE não faz nenhuma validação no navegador: seu schema está vazio.
// Tudo é enviado ao servidor, que responde, em caso de recusa, com um erro
// JSON (convertido em um ApiError pelo interceptor):
//
//   400 { key: 'ERRORS.INVALID_FORM', fields: { lastName: 'ERRORS.LASTNAME_REQUIRED' } }
//   409 { key: 'ERRORS.PATIENT_ALREADY_EXISTS', fields: { firstName: ..., lastName: ... } }
//   409 { key: 'ERRORS.CONCURRENT_UPDATE' }         (bloqueio otimista)
//
// submit(form, action): a action envia o modelo ao servidor. Se ela
// retornar ERRORS (um array), os signal forms os associam
// aos campos designados (fieldTree): eles aparecem em f.lastName().errors(),
// como os erros de um validador. Sua mensagem é aqui a CHAVE enviada pelo
// servidor, que o template traduz (ngx-translate). Experimente: um sobrenome
// vazio, um título "Dr", um homônimo (Alice DURAND), um login já em uso
// (martin), o mesmo registro modificado em duas abas...
// =========================================================================

import { Component, computed, inject, signal } from '@angular/core';
import { FormField, form, submit, ValidationError } from '@angular/forms/signals';
import { TranslatePipe, TranslateService } from '@ngx-translate/core';
import { ApiService } from '../../../common/api.service';
import { LoginBar } from '../../../common/login-bar';
import { ApiError } from '../../../common/api-error';
import { Person, User } from '../../../common/types';
// o diálogo de confirmação do exemplo 16, reutilizado como está
import { ConfirmationService } from '../../../16-confirmation/src/app/confirmation/confirmation.service';
import { ConfirmDialog } from '../../../16-confirmation/src/app/confirmation/confirm-dialog';

interface ClientInput {
  title: string;
  firstName: string;
  lastName: string;
  login: string;
  password: string;
  version: number;
}

const EMPTY: ClientInput = { title: 'Mr', firstName: '', lastName: '', login: '', password: '', version: 0 };

@Component({
  selector: 'app-root',
  imports: [FormField, TranslatePipe, LoginBar, ConfirmDialog],
  templateUrl: './app.html',
})
export class App {
  private readonly api = inject(ApiService);
  private readonly confirmation = inject(ConfirmationService);
  protected readonly translate = inject(TranslateService);

  protected readonly user = signal<User | null>(null);
  protected readonly clients = signal<Person[]>([]);
  protected readonly isAdmin = computed(() => this.user()?.role === 'ADMIN');

  // o formulário: editedId === null -> criação, caso contrário atualização
  protected readonly editedId = signal<number | null>(null);
  protected readonly model = signal<ClientInput>({ ...EMPTY });
  protected readonly f = form(this.model); // nenhum validador: o servidor valida
  // a mensagem geral do último erro e a de um sucesso
  protected readonly pageError = signal<ApiError | null>(null);
  protected readonly success = signal<{ key: string; params: Record<string, string> } | null>(null);

  protected async onUserChange(u: User | null): Promise<void> {
    this.user.set(u);
    this.clients.set([]);
    if (u?.role === 'ADMIN') {
      await this.reload();
    }
  }

  private async reload(): Promise<void> {
    this.clients.set(await this.api.get<Person[]>('/clients'));
  }

  protected newForm(): void {
    this.editedId.set(null);
    this.model.set({ ...EMPTY });
    this.f().reset(); // o estado dos campos (e seus erros) parte do zero
    this.clearMessages();
  }

  protected edit(c: Person): void {
    this.editedId.set(c.id);
    // a VERSÃO lida com o registro volta com a atualização: se alguém
    // modificou o registro nesse meio tempo, o servidor responderá 409 CONCURRENT_UPDATE
    this.model.set({ title: c.title, firstName: c.firstName, lastName: c.lastName, login: c.login ?? '', password: '', version: c.version });
    this.f().reset();
    this.clearMessages();
  }

  private clearMessages(): void {
    this.pageError.set(null);
    this.success.set(null);
  }

  // o erro de CADA campo, retornado pelo servidor -> um erro do formulário
  private fieldErrors(e: ApiError): ValidationError.WithOptionalFieldTree[] {
    const fields = Object.keys(this.model()) as (keyof ClientInput)[];
    return Object.entries(e.fields)
      .filter(([field]) => fields.includes(field as keyof ClientInput))
      .map(([field, key]) => ({ kind: 'server', message: key, fieldTree: this.f[field as keyof ClientInput] }));
  }

  protected async save(event: SubmitEvent): Promise<void> {
    event.preventDefault();
    this.clearMessages();
    const id = this.editedId();
    await submit(this.f, async () => {
      try {
        const body = this.model();
        const client =
          id === null
            ? await this.api.post<Person>('/clients', body)
            : await this.api.put<Person>(`/clients/${id}`, body);
        this.newForm(); // formulário limpo, mensagens limpas...
        // ... exceto esta
        this.success.set({
          key: id === null ? 'FORM.ADDED' : 'FORM.UPDATED',
          params: { name: `${client.firstName} ${client.lastName}`, login: client.login ?? '' },
        });
        await this.reload();
        return null;
      } catch (e) {
        if (!(e instanceof ApiError)) {
          throw e;
        }
        this.pageError.set(e); // a mensagem geral
        return this.fieldErrors(e); // o erro de cada campo
      }
    });
  }

  protected async remove(c: Person): Promise<void> {
    const name = `${c.firstName} ${c.lastName}`;
    if (!(await this.confirmation.confirm(this.translate.instant('FORM.DELETE_CONFIRM', { name })))) {
      return;
    }
    this.clearMessages();
    try {
      await this.api.delete(`/clients/${c.id}`);
      this.success.set({ key: 'FORM.DELETED', params: { name } });
    } catch (e) {
      this.pageError.set(e instanceof ApiError ? e : null);
    }
    await this.reload();
  }
}
